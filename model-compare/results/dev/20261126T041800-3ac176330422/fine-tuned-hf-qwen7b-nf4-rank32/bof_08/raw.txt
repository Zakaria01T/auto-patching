void copy_data(const char *src, size_t len) {
    char buf[16];
    if (len > sizeof(buf))
        len = sizeof(buf);
    memcpy(buf, src, len);
    buf[len] = '\0';
    printf("%.*s\n", (int)len, buf);
}
