```c
void copy_data(const char *src, size_t len) {
    char buf[16];
    if (len >= sizeof(buf)) {
        len = sizeof(buf) - 1;
    }
    memcpy(buf, src, len);
    buf[len] = '\0';
    printf("%.*s\n", (int)len, buf);
}
```

EXPLANATION: Ensured that `len` does not exceed the buffer size, preventing overflow, and guaranteed null termination.
