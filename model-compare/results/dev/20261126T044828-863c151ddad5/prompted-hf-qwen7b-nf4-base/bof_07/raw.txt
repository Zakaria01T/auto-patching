void full_name(const char *first, const char *last) {
    char name[20];
    if (strlen(first) + strlen(last) + 2 <= sizeof(name)) {
        strcpy(name, first);
        strcat(name, " ");
        strcat(name, last);
        puts(name);
    } else {
        // Handle error or truncate the string
        fprintf(stderr, "Name too long\n");
    }
}
